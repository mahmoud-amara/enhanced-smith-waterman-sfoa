# helper: rounded box
def rbox(ax, x, y, w, h, text, fc=IEEE_LTBLUE, ec=IEEE_BLUE,
         fontsize=8, txt_color='white', lw=1.2, zorder=3):
    box = FancyBboxPatch((x, y), w, h,
                         boxstyle="round,pad=0.12", linewidth=lw,
                         edgecolor=ec, facecolor=fc, zorder=zorder,
                         transform=ax.transData)
    ax.add_patch(box)
    ax.text(x + w/2, y + h/2, text, ha='center', va='center',
            fontsize=fontsize, fontweight='bold', color=txt_color,
            zorder=zorder+1, transform=ax.transData)
    return box


def arrow_h(ax, x1, y, x2, color=IEEE_BLUE, lw=1.5):
    ax.annotate('', xy=(x2, y), xytext=(x1, y),
                arrowprops=dict(arrowstyle='->', color=color, lw=lw),
                zorder=2)


def arrow_v(ax, x, y1, y2, color=IEEE_BLUE, lw=1.5):
    ax.annotate('', xy=(x, y2), xytext=(x, y1),
                arrowprops=dict(arrowstyle='->', color=color, lw=lw),
                zorder=2)

def figure2_traditional_sw():

    fig, ax = plt.subplots(figsize=(7, 6))

    n, m = 20, 20
    # Draw full DP grid
    for i in range(n+1):

                                 edgecolor=IEEE_LTGRAY, facecolor='#E3F2FD',
                                 zorder=1)
            ax.add_patch(rect)

    # Traceback path (diagonal)
    tb = [(1,1),(2,2),(3,3),(4,4),(5,4),(6,5),(7,6),(8,7),(9,8),
          (10,9),(11,10),(12,11),(13,12),(14,13),(15,14),(16,15),

    for i, j in tb:
        rect = plt.Rectangle((j, n-i), 1, 1, linewidth=0.4,
                              edgecolor=IEEE_BLUE, facecolor=IEEE_CYAN,
                              alpha=0.7, zorder=2)

    ax.set_ylim(0, n+1)

def figure3_adaptive_banding():
    print("[Fig 3] SW + Adaptive Banding …")
    fig, ax = plt.subplots(figsize=(7, 6))

    n = 20
    band = 4  # visual band width

    for i in range(n+1):
        for j in range(n+1):
            in_band = abs(i - j) <= band
            if in_band:
                rect = plt.Rectangle((j, n-i), 1, 1, linewidth=0.4,
                                     edgecolor=IEEE_BLUE, facecolor='#B3E5FC',
                                     zorder=2)
            else:
                rect = plt.Rectangle((j, n-i), 1, 1, linewidth=0.3,
                                     edgecolor='#E0E0E0', facecolor='#F5F5F5',
                                     alpha=0.5, zorder=1)
            ax.add_patch(rect)

    # Band boundary lines
    xs = np.linspace(0, n+1, 100)
    ax.plot(xs, [n+1 - (x - band) for x in xs], '--', color=IEEE_ORANGE,
            lw=1.5, zorder=3, label='Band Boundary')
    ax.plot(xs, [n+1 - (x + band) for x in xs], '--', color=IEEE_ORANGE,
            lw=1.5, zorder=3)

    ax.set_xlim(0, n+1)
    ax.set_ylim(0, n+1)
    ax.set_aspect('equal')
    ax.set_xlabel('Sequence 2', fontsize=11, color=IEEE_BLUE)
    ax.set_ylabel('Sequence 1', fontsize=11, color=IEEE_BLUE)
    ax.set_title('Smith-Waterman with Adaptive Banding', fontsize=13,
                 fontweight='bold', color=IEEE_BLUE, pad=12)

    # Annotations
    ax.annotate('Reduced Search\nSpace', xy=(15, 15), fontsize=9,
                fontweight='bold', color=IEEE_GREEN, ha='center',
                bbox=dict(boxstyle='round,pad=0.3', fc='#E8F5E9', ec=IEEE_GREEN))

    ax.text(3, 18, r'$\mathcal{O}(n \times b)$ Complexity',
            fontsize=11, fontweight='bold', color=IEEE_CYAN,
            bbox=dict(boxstyle='round,pad=0.3', fc='white', ec=IEEE_CYAN))

    ax.text(16, 3, 'Adaptive\nCorridor', fontsize=9, color=IEEE_ORANGE,
            ha='center', fontstyle='italic', fontweight='bold')

    # Legend
    legend_els = [
        mpatches.Patch(facecolor='#B3E5FC', edgecolor=IEEE_BLUE, label='Active Band Cells'),
        mpatches.Patch(facecolor='#F5F5F5', edgecolor='#E0E0E0', label='Skipped Cells'),
        Line2D([0],[0], color=IEEE_ORANGE, ls='--', lw=1.5, label='Band Boundary'),
    ]
    ax.legend(handles=legend_els, loc='upper right', fontsize=8,
              framealpha=0.95, edgecolor=IEEE_LTGRAY)
    ax.set_xticks([])
    ax.set_yticks([])

    # band formula
    ax.text(n/2+0.5, -1.2,
            r'$b = b_{max} \times (1 - similarity) + b_{min}$',
            ha='center', fontsize=10, color=IEEE_BLUE,
            bbox=dict(boxstyle='round,pad=0.3', fc='#E3F2FD', ec=IEEE_BLUE))

    fig.text(0.5, -0.05,
             "Fig. 3. Adaptive banding restricts computation to a diagonal corridor.\n"
             "Band width adapts inversely to sequence similarity.",
             ha='center', fontsize=9, color=IEEE_DKGRAY, fontstyle='italic')

    save(fig, 'fig3_adaptive_banding.png')

def figure5_comparative():
    print("[Fig 5] Comparative Methodology …")
    fig, axes = plt.subplots(1, 4, figsize=(14, 4.5))

    n = 15
    titles = [
        "(a) Traditional SW",
        "(b) Banded SW",
        "(c) Adaptive SW",
        "(d) Proposed (PF+SFOA)",
    ]
    bands = [None, 3, 5, 4]  # None = full, then fixed, adaptive, optimized
    colors_active = ['#E3F2FD', '#B3E5FC', '#80DEEA', '#80CBC4']
    colors_edge   = [IEEE_LTGRAY, IEEE_BLUE, IEEE_CYAN, '#00695C']

    for idx, ax in enumerate(axes):
        b = bands[idx]
        active_count = 0
        total_count = (n+1)**2

        for i in range(n+1):
            for j in range(n+1):
                if b is None:
                    active = True
                else:
                    active = abs(i - j) <= b
                if active:
                    active_count += 1
                    rect = plt.Rectangle((j, n-i), 1, 1, lw=0.3,
                                         ec=colors_edge[idx],
                                         fc=colors_active[idx], zorder=2)
                else:
                    rect = plt.Rectangle((j, n-i), 1, 1, lw=0.2,
                                         ec='#E0E0E0', fc='#FAFAFA',
                                         alpha=0.4, zorder=1)
                ax.add_patch(rect)

        ax.set_xlim(0, n+1)
        ax.set_ylim(0, n+1)
        ax.set_aspect('equal')
        ax.set_title(titles[idx], fontsize=9, fontweight='bold', color=IEEE_BLUE, pad=8)
        ax.set_xticks([])
        ax.set_yticks([])

        pct = active_count / total_count * 100
        ax.text(n/2+0.5, -0.8, f"Cells: {pct:.0f}%",
                ha='center', fontsize=8, fontweight='bold', color=IEEE_DKGRAY)

        if idx == 0:
            ax.text(n/2+0.5, -1.8, r"$\mathcal{O}(n \times m)$",
                    ha='center', fontsize=9, color=IEEE_RED)
        elif idx >= 1:
            ax.text(n/2+0.5, -1.8, r"$\mathcal{O}(n \times b)$",
                    ha='center', fontsize=9, color=IEEE_CYAN)

    fig.suptitle("Comparative Methodology — Matrix Computation Reduction",
                 fontsize=13, fontweight='bold', color=IEEE_BLUE, y=1.02)

    fig.text(0.5, -0.08,
             "Fig. 5. Side-by-side comparison of DP matrix utilization across methods.\n"
             "(d) Proposed framework achieves optimized band width via SFOA + Pufferfish indexing.",
             ha='center', fontsize=9, color=IEEE_DKGRAY, fontstyle='italic')

    plt.tight_layout()
    save(fig, 'fig5_comparative_methodology.png')
